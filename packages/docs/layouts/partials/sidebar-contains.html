{{- /* Returns true if any of the sidebar items (recursively) link to the given page. Params: items, page */ -}} {{- $page := .page -}} {{- $found := false -}} {{- range .items -}} {{- if not $found -}} {{- if eq .type "doc" -}} {{- with site.GetPage .id }}{{ if eq .RelPermalink $page.RelPermalink }}{{ $found = true }}{{ end }}{{ end -}} {{- else if eq .type "category" -}} {{- with .id }}{{ with site.GetPage . }}{{ if eq .RelPermalink $page.RelPermalink }}{{ $found = true }}{{ end }}{{ end }}{{ end -}} {{- if not $found }}{{ $found = partial "sidebar-contains" (dict "items" .items "page" $page) }}{{ end -}} {{- else if eq .type "autogenerated" -}} {{- /* Anything under the listed section, at any depth */ -}} {{- with site.GetPage .dir }}{{ if and (ne .RelPermalink $page.RelPermalink) (hasPrefix $page.RelPermalink .RelPermalink) }}{{ $found = true }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $found -}}
