{{- /* The code shown by the code-example shortcode, as dict "code" (the text) and "highlight" (line numbers to highlight): lines from a repo file (mounted under assets/code/), limited to the named blocks if any, with marker comments and the common indentation removed. Params: file, blocks, position (the shortcode's .Position, for errors). */ -}} {{- $file := .file -}} {{- $resource := resources.Get (printf "code/%s" $file) -}} {{- if not $resource -}} {{- errorf "%s: code-example file not found: %s" $.position $file -}} {{- end -}} {{- $markerPattern := `^\s*//\s*(start|end)-block\s+([A-Za-z_-]*)` -}} {{- $lines := split (strings.TrimRight "\n" $resource.Content) "\n" -}} {{- $selected := slice -}} {{- with .blocks -}} {{- range $name := split . "," -}} {{- $name = trim $name " " -}} {{- $inBlock := false -}} {{- $found := false -}} {{- range $lines -}} {{- $marker := findRESubmatch $markerPattern . 1 -}} {{- if $marker -}} {{- if eq (index $marker 0 2) $name -}} {{- $inBlock = eq (index $marker 0 1) "start" -}} {{- $found = true -}} {{- end -}} {{- else if $inBlock -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- errorf "%s: no block %q in %s" $.position $name $file -}} {{- end -}} {{- end -}} {{- else -}} {{- range $lines -}} {{- if not (findRE $markerPattern .) -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Remove the common indentation */ -}} {{- $indent := 1000 -}} {{- range $selected -}} {{- if trim . " \t" -}} {{- $indent = int (math.Min $indent (len (index (findRE `^[ \t]*` . 1) 0))) -}} {{- end -}} {{- end -}} {{- /* Docusaurus's magic comments (// or # highlight-next-line, highlight-start, highlight-end) mark lines to highlight; drop them and collect the line numbers they mark, counted without them */ -}} {{- $code := slice -}} {{- $highlight := slice -}} {{- $next := false -}} {{- $inRange := false -}} {{- range $selected -}} {{- $line := cond (gt (len .) $indent) (substr . $indent) (trim . " \t") -}} {{- $magic := index (findRESubmatch `^\s*(?://|#)\s*highlight-(next-line|start|end)\s*$` $line 1) 0 -}} {{- if $magic -}} {{- $kind := index $magic 1 -}} {{- if eq $kind "next-line" }}{{ $next = true }}{{ else }}{{ $inRange = eq $kind "start" }}{{ end -}} {{- continue -}} {{- end -}} {{- $code = $code | append $line -}} {{- if or $next $inRange }}{{ $highlight = $highlight | append (string (len $code)) }}{{ end -}} {{- $next = false -}} {{- end -}} {{- return dict "code" (delimit $code "\n") "highlight" $highlight -}}
