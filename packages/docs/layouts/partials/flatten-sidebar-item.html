{{- /* Flattens one data/sidebar.yml item into its pages, in sidebar order (for the doc footer's previous/next links) */ -}} {{- $item := . -}} {{- $pages := slice -}} {{- if eq $item.type "doc" -}} {{- with site.GetPage $item.id }}{{ $pages = $pages | append . }}{{ end -}} {{- else if eq $item.type "category" -}} {{- with $item.id }}{{ with site.GetPage . }}{{ $pages = $pages | append . }}{{ end }}{{ end -}} {{- range $item.items }}{{ $pages = $pages | append (partial "flatten-sidebar-item" .) }}{{ end -}} {{- else if eq $item.type "autogenerated" -}} {{- with site.GetPage $item.dir -}} {{- range .Pages.ByWeight -}} {{- $pages = $pages | append . -}} {{- if and .IsSection (gt (len .Pages) 0) }}{{ $pages = $pages | append (partial "flatten-sidebar-item" (dict "type" "autogenerated" "dir" .Path)) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $pages -}}
