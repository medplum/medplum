{{- /* Renders code from a repo file (mounted under assets/code/), so docs show the type-checked example source. Same semantics as the Docusaurus MedplumCodeBlock: `blocks` selects the lines between `// start-block ` and `// end-block ` comments, marker comments are dropped, and the common indentation is removed. Params: file (repo-relative path), language, blocks (comma-separated), title, lineNumbers, highlight ("1,4-6") */ -}} {{- $file := .Get "file" -}} {{- $resource := resources.Get (printf "code/%s" $file) -}} {{- if not $resource -}} {{- errorf "%s: code-example file not found: %s" .Position $file -}} {{- end -}} {{- $markerPattern := `^\s*//\s*(start|end)-block\s+([A-Za-z_-]*)` -}} {{- $lines := split (strings.TrimRight "\n" $resource.Content) "\n" -}} {{- $selected := slice -}} {{- with .Get "blocks" -}} {{- range $name := split . "," -}} {{- $name = trim $name " " -}} {{- $inBlock := false -}} {{- $found := false -}} {{- range $lines -}} {{- $marker := findRESubmatch $markerPattern . 1 -}} {{- if $marker -}} {{- if eq (index $marker 0 2) $name -}} {{- $inBlock = eq (index $marker 0 1) "start" -}} {{- $found = true -}} {{- end -}} {{- else if $inBlock -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- errorf "%s: no block %q in %s" $.Position $name $file -}} {{- end -}} {{- end -}} {{- else -}} {{- range $lines -}} {{- if not (findRE $markerPattern .) -}} {{- $selected = $selected | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Remove the common indentation */ -}} {{- $indent := 1000 -}} {{- range $selected -}} {{- if trim . " \t" -}} {{- $indent = int (math.Min $indent (len (index (findRE `^[ \t]*` . 1) 0))) -}} {{- end -}} {{- end -}} {{- $code := slice -}} {{- range $selected -}} {{- $code = $code | append (cond (gt (len .) $indent) (substr . $indent) (trim . " \t")) -}} {{- end -}} {{- $options := dict -}} {{- with .Get "highlight" -}} {{- $options = merge $options (dict "hl_lines" (replace . "," " ")) -}} {{- end -}} {{- if eq (.Get "lineNumbers") "true" -}} {{- $options = merge $options (dict "lineNos" "table") -}} {{- end -}} {{- with .Get "title" -}}
{{ . }}
{{- end -}} {{- highlight (delimit $code "\n") (.Get "language" | default "text") $options -}}
